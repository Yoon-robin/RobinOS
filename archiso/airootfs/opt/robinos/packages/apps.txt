# RobinOS apps for the installed system only. The live ISO leaves them out to
# stay under GitHub's 2 GiB release asset limit (archiso/packages.x86_64);
# installer/robin-install and scripts/post-install.sh install them.

# Printing: CUPS finds network printers through Avahi (name.local), and the
# printer settings app adds them ("장치 및 프린터" in the launcher)
cups
nss-mdns
system-config-printer

# Office: LibreOffice's stable branch with the Korean interface and help
# ("워드", "엑셀", "파워포인트" in the launcher). About 150 MiB to download.
libreoffice-still
libreoffice-still-ko

# Disks: format a USB stick, check partitions ("디스크 관리" in the launcher)
gnome-disk-utility
# What fills the disk, like Windows' storage settings ("저장소", "디스크 정리")
baobab
# Backups of personal files to a USB disk or a network folder (Deja Dup), like
# Windows' File History ("백업", "파일 히스토리" in the launcher). Snapshots only
# cover the system (docs/recovery.md "스냅샷과 백업")
deja-dup

# App store for Flathub apps (flatpak ships the Flathub remote). Without
# PackageKit it never touches pacman; robinctl update updates its apps too.
gnome-software
# Firmware updates (LVFS) for the PC's BIOS, SSD and devices; gnome-software
# lists them under its updates when fwupd is installed
fwupd

# Video and music: GNOME's players ("동영상 재생 앱", "오디오 재생기"; "미디어
# 플레이어", "그루브 음악" in the launcher). gst-libav plays H.264/AAC (MP4) through
# ffmpeg, which the system already has. desktop/mime/mimeapps.list makes them the
# default for video and audio files.
showtime
decibels
gst-libav
